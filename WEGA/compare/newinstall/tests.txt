# The cases of the corpus of new.install (compare/newinstall/corpus.py); the format: host/recon/casefile.py. Every case runs as root with
# the clock at 600000000 and starts with four file systems of 400 blocks as images (usr, root, tmp, z: see corpus.py), the layer's
# lost+found of / and /usr and the man pages of man0 to man5 as links. The answers of the operator are the standard input (gets). The
# original mkfs, labelit, dd, mfs, makelost+found and unpack run in it. Where an expectation comes from: src = src/cmd/new.install.sh
# (a csh script, run by /bin/csh); wegafs.py info reads the super block (the name written by labelit, the size made by mkfs).

== no
# src: the answer n: the file systems /tmp and /z are not made again; root and /usr are named (labelit: root, /usr), the boot block
# (pb.image, 464 bytes) is written to /dev/rmd0 (the first block of the image of /usr), /etc/mfs mounts, lost+found of / and /usr
-- a
n
-- case
step: /etc/new.install
stdin: a
host: for f in usr root tmp z; do python3 $WEGA_HOST/tools/wegafs.py info $f.img | grep "fname" | tr '\n' ' '; echo; done; cmp -n 464 usr.img $WEGA_ROOT/pb.image && echo boot-same
expect: exit 0
expect: output contains 'Benennen der Dateisysteme'
expect: output contains 'NEW fsname = root, NEW volname = 1 -- DEL if wrong!!'
expect: output contains 'NEW fsname = /usr, NEW volname = 1 -- DEL if wrong!!'
expect: output contains 'dd if=/pb.image of=/dev/rmd0 conv=sync count=1'
expect: output contains 'make lost+found runs about 3 minutes for /usr'
expect: output contains 'fname  /usr'
expect: output contains 'boot-same'

== yes-small

expect: output contains 'make lost+found runs about 3 minutes for /tmp'
expect: output contains 'make lost+found runs about 3 minutes for /z'
expect: output contains 'fsize  300'
expect: output contains 'fsize  500'

== packed-pages
# src: the manual pages are unpacked (/usr/bin/unpack) at the end. After /etc/mfs /usr is the mounted image: the page packed by the
# original pack is put into the image of /usr (man/man0/INTRO.0.z, and bin/unpack: the mounted /usr has nothing else) and unpack finds it (it is unpacked in the mounted file system, which
# is not kept: only the message shows it); the other directories of the pages are not there, unpack says so on the diagnostic output
-- a
n
-- case
host: cp $WEGA_ROOT/usr/man/man0/INTRO.0 INTRO.0
step: /usr/bin/pack INTRO.0
host: python3 $WEGA_HOST/tools/wegafs.py mkdir usr.img /bin 755 0 0 500000000; python3 $WEGA_HOST/tools/wegafs.py put usr.img /bin/unpack $WEGA_ROOT/usr/bin/unpack 755 0 0 500000000; python3 $WEGA_HOST/tools/wegafs.py mkdir usr.img /man 755 0 0 500000000; python3 $WEGA_HOST/tools/wegafs.py mkdir usr.img /man/man0 755 0 0 500000000; python3 $WEGA_HOST/tools/wegafs.py put usr.img /man/man0/INTRO.0.z INTRO.0.z 644 0 0 500000000
step: /etc/new.install
stdin: a
expect: output contains 'Expandieren der Handbucheintraege in man0'
host: find $Z8K_MOUNTS* -name 'INTRO*' | sed 's#.*/man/#man/#'
expect: output contains 'man/man0/INTRO.0'
