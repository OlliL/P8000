# The cases of the corpus of rm (compare/rm/corpus.py); the format: host/recon/casefile.py. Compared: the output, the exit
# status and the directory afterwards. Where an expectation comes from: man = /usr/man/man1/rm.1, src = src/cmd/rm.c.
# Without a terminal on the standard input rm sets -f itself (src: !isatty(0)); the emulator has no terminal, so every
# case runs as with -f: "x nonexistent" and the question for a file without write permission ("rm: x 444 mode") need the
# terminal model (increment 120, not yet).

# ---- the public interface (man page) ----------------------------------------------------------------------------

== files
# man: rm removes the entries of the files named
-- a
-- b
-- c
-- case
step: rm a b
expect: exit 0
expect: missing a
expect: missing b
expect: file c contains ''

== last-link
# man: the file goes when its last link goes: another link keeps it
-- a
data
-- case
host: ln a b
step: rm a
expect: missing a
expect: file b contains data

== no-write-permission
# man: removing needs write permission on the directory, not on the file (-f, no terminal: no question)
-- ro
x
-- case
host: chmod 444 ro
step: rm ro
expect: missing ro

== no-dir-permission
# man: without write permission on the directory the entry stays (with -f no message, src: no message without -i)
-- d/f
x
-- case
host: chmod 555 d
step: rm d/f
host: chmod 755 d
expect: file d/f contains x
expect: exit 0

== directory
# man: a directory is an error without -r
-- d/f
-- case
step: rm d
expect: output is 'rm: d directory\n'
expect: exit 1

== recursive
# man: -r removes the whole contents of the directory and the directory itself (src: also entries starting with a point)
-- d/f
-- d/.hidden
-- d/sub/g
-- d/sub/deeper/h
-- keep
-- case
step: rm -r d
expect: missing d
expect: exit 0

== force
# man: -f asks nothing; a name that does not exist is no error with -f
-- a
-- case
step: rm -f a nosuch
expect: missing a
expect: exit 0

== interactive
# man: -i asks for every file; a line starting with y removes it (src: also Y, blanks and tabs before it: \040 \011)
-- a
-- b
-- c
-- d
-- e
-- answers
y
n
Y
\040\040\011y

-- case
step: rm -i a b c d e
stdin: answers
expect: output is 'a: b: c: d: e: '
expect: missing a
expect: file b contains ''
expect: missing c
expect: missing d
expect: file e contains ''

== interactive-recursive
# man: -i with -r asks for every directory to be examined and every file; src: the directory named itself is not asked
# before it is examined, only before rmdir; the answers go in the order of the questions (the order of the entries is the
# one of the directory: g is removed either way)
-- d/f
-- d/sub/g
-- answers
y
y
y
n
-- case
step: rm -ri d
stdin: answers
expect: output contains 'directory d/sub: '
expect: missing d/sub/g

== dotdot
# man (DIAGNOSTIK): . and .. cannot be removed; src: ".." is refused with a message on the standard error
-- case
step: rm ..
expect: output contains "cannot remove `..'"

== recursive-dot
# src: rm -r . removes everything below the current directory, but not "." itself (rmdir is not run for a name . or ..)
-- a
-- d/f
-- case
step: rm -r .
expect: missing a
expect: missing d

# src: rm without a file prints the usage too, but ends with 0
-- case

== options-only

y

== exit-count
# src: the exit status counts the errors (here two directories without -r)
-- d1/f
-- d2/f
-- case
step: rm d1 d2
expect: exit 2

== rmdir-status
# src: rm -r adds the status of rmdir as wait() gives it (rmdir exits with 2: 512) to the count of errors: one error of
# rm (the directory without write permission) and the failed rmdir give 513, of which exit keeps 1
-- d/sub/f
-- case
host: chmod 555 d/sub
step: rm -r d
host: chmod 755 d/sub
expect: exit 1
expect: output contains 'rmdir: d not empty'

== interactive-unlink-error

stdin: answers
host: chmod 755 d
expect: output contains 'rm: cannot remove d/f'
expect: exit 1

== interact-all

-- a
-- b
-- c
-- answers
y
n
y
-- case
step: rm -i
stdin: answers
expect: output contains '[removed]'
expect: output contains '[untouched]'

== interact-eof
# src: rm -i without a file, the standard input at its end: every entry is left
-- a
-- b
-- case

expect: file a contains ''
expect: exit 0

== long-name
# src: names of 14 characters (the length of a directory entry) below a directory removed with -r
-- d/abcdefghijklmn
-- d/abcdefghijklm
-- case

== empty-answer

# ---- the users and the terminal (user:, terminal: console) ------------------------------------------------------------
# On a terminal rm asks before it removes a file the user may not write (src: access with the real user); root may
# write every file (fio.c access): no question. Removing a name needs write permission on the directory.

== user-question
# src: "rm: b 444 mode " and an answer; n keeps the file
-- b
x
-- n
n
-- case
terminal: console
user: 100 10
host: chmod 444 b
step: rm b
stdin: n
host: ls
expect: output contains 'rm: b 444 mode '
expect: output contains 'b'

== user-question-yes
# src: y removes it
-- b
x
-- y
y
-- case
terminal: console
user: 100 10
host: chmod 444 b
step: rm b
stdin: y
expect: output contains 'rm: b 444 mode '
expect: missing b

== user-root-no-question
# fio.c access: root may write it: no question
-- b
x
-- case

expect: output is ''
expect: missing b

== user-unwritable-dir
# src: the unlink fails (the directory is not writable for the user): perror; root removes it
-- case
terminal: console
user: 100 10
host: mkdir d && echo x > d/a && chmod 555 d
step: rm d/a
host: chmod 755 d; ls d
expect: output contains 'rm: cannot remove d/a: Permission denied'
expect: output contains 'step 1 rm: exit=1'

== user-root-unwritable-dir
# fio.c access: root may remove from it
-- case

step: rm d/a
host: chmod 755 d
expect: missing d/a

== user-r-unwritable
# src: rm -r of a directory the user may not write: "d not changed" (on a terminal; without one rm is quiet as with -f)
-- case

step: rm -r d
host: chmod 755 d
expect: output contains 'd not changed'

== user-root-r-unwritable

-- case

step: rm -r d
