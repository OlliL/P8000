# The cases of the corpus of adduser (compare/adduser/corpus.py); the format: host/recon/casefile.py. Every case runs as root with the
# clock at 600000000 and the console as terminal. The standard input holds the answers of the operator (read by gets) and at the end the
# password twice (read by passwd; it must have 6 characters). /etc/passwd, /etc/group, /etc/whois are copies in the layer; home
# directories are made in usr/home of the layer, the mail files in usr/spool/mail. The salt of the password is made of the process
# number and the time: both are fixed. adduser has no man page in the tree: src = src/cmd/adduser.sh (a csh script).

== add
# src: a new user: the next free uid (below 20 the loop takes the first free number above the ones in use), the group number of the
# default group (other), the home directory is made, the line goes to /etc/passwd with the password from passwd, the user is added
# to the other group, whois gets a line, the mail file is made (owner and group by chmog)
-- in
newguy
/usr/home/newguy


adm

yes
secret1
secret1
-- case
step: /etc/adduser
stdin: in
host: grep "^newguy" $Z8K_OVERLAY/etc/passwd $Z8K_OVERLAY/etc/whois | sed "s#$Z8K_OVERLAY##"; grep "^adm" $Z8K_OVERLAY/etc/group; stat -c '%a %n' $Z8K_OVERLAY/usr/home/newguy $Z8K_OVERLAY/usr/spool/mail/newguy | sed "s#$Z8K_OVERLAY##"
expect: exit 0
expect: output contains 'Uid:             25'
expect: output contains 'newguy: Added'
expect: output contains '/etc/passwd:newguy:'
expect: output contains ':25:1::/usr/home/newguy:/bin/csh'
expect: output contains '/etc/whois:newguy:::::'
expect: output contains 'adm::4:newguy'
