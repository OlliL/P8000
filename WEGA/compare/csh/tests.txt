== echo
# sh.func.c echo: the words with one blank; -n: no newline; \t \n \\ \b \f \r and \0nnn (three octal digits at most);
# an escape suppresses the blank behind its word; \c ends the output
-- case
step: csh t.in
expect: output is 'hello world\nno newline\na	bc\noct1 z\nxafter\n'
-- t.in
echo hello   world
echo -n no newline
echo
echo 'a\tb' c
echo 'oct\0101' z
echo 'x\c' y
echo after

== while

== switch
# sh.func.c doswitch, search ZCASE ZDEFAULT
-- case
step: csh t.in
expect: exit 1
expect: output is '-- stderr\nVariable syntax.\n'
-- t.in
foreach w (abc xyz 12)
switch ($w)
case a*:
	echo $w: a
	breaksw
case [0-9]*:
	echo $w: digit
	breaksw
default:
	echo $w: default
endsw
end

== goto
