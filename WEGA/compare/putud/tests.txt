# The cases of the corpus of putud (compare/putud/corpus.py); the format: host/recon/casefile.py. A case runs on the
# console (terminal: console) with a UDOS disk in drive 0 (an image made by host/tools/udosfs.py) and the model of the
# 8-bit part behind /dev/ud (host/z8k-emu/models/ud8.py). Compared: the output, what the 8-bit part prints on its console
# ("8-bit console:" at the end of the record), the disk image and the files. udosfs.py ls shows the directory.
# Where an expectation comes from: man = /usr/man/man1/putud.1, src = src/cmd/putud.c, k = the coupling software
# (UDOS/WEGA/K.ACT.S: request L).

# ---- the public interface (man page) ----------------------------------------------------------------------------

== ascii
# man: putud writes a file in UDOS format onto the floppy; src: an ASCII file, newlines become returns; k: type A, record
# length 200H, the end mark FFH, the last record filled with FFH
-- text
first line
second line
-- case
terminal: console
device: drive0 disk.img
device: ud
host: python3 $WEGA_HOST/tools/udosfs.py format disk.img
step: putud text
host: python3 $WEGA_HOST/tools/udosfs.py ls disk.img
host: python3 $WEGA_HOST/tools/udosfs.py cat disk.img text | od -c | head -3
expect: output contains 'text                             A0   -          1 0200    23'
expect: output contains 'f   i   r   s   t       l   i   n   e  \r'

== binary

== replace
# k: an existing file is opened for OUTPUT: its records are deleted and written again; measured in MAME: it takes the
# record length 200H of the coupling software, its date of creation stays
-- text
new contents
-- case

host: python3 $WEGA_HOST/tools/udosfs.py format disk.img && printf 'old\r' > old && python3 $WEGA_HOST/tools/udosfs.py put disk.img text A 100 old 870101

== disk-full
# k: no room on the disk: DISK FULL D3 (the model's writer; the disk is filled first: 2551 sectors are free after
# formatting, BIG takes 2528 records, its descriptor and 21 pointer sectors; one sector is left, a file needs three)
-- text
x
-- case
terminal: console
device: drive0 disk.img
device: ud
host: python3 $WEGA_HOST/tools/udosfs.py format disk.img && head -c 647168 /dev/zero > big && python3 $WEGA_HOST/tools/udosfs.py put disk.img BIG B 100 big
step: putud text
expect: output contains 'Error D3'

== descriptor
# src: -f sends a UDOS file with its descriptor (k: the first 80H bytes: 6 times MAGIC 0D607H and the attributes);
# a file made by getud -f -b of a procedure file, sent back under another name; -b too: without it putud turns every
# newline byte of the data into a return, also with -f
-- case
terminal: console
device: drive0 disk.img
device: ud
host: python3 $WEGA_HOST/tools/udosfs.py format disk.img && python3 -c "import sys; sys.stdout.buffer.write(bytes(i * 37 % 256 for i in range(700)))" > p && python3 $WEGA_HOST/tools/udosfs.py put disk.img PROG P 100 p
step: getud -f -b PROG
host: mv PROG COPY
step: putud -f -b COPY
host: python3 $WEGA_HOST/tools/udosfs.py ls disk.img
host: python3 $WEGA_HOST/tools/udosfs.py cat disk.img COPY | cmp - p && echo same data
expect: output contains 'COPY                             P0'
expect: output contains 'same data'

== descriptor-invalid
# k (WRTDES): -f with a file that does not begin with MAGIC: "INVALID DESCRIPTOR" on the 8-bit console when its first
# 80H bytes have come (a shorter file never fills the descriptor: not a case, what then happens is not known)
-- case
host: seq 1 100 > text
terminal: console
device: drive0 disk.img
device: ud
host: python3 $WEGA_HOST/tools/udosfs.py format disk.img
step: putud -f text
expect: output contains 'INVALID DESCRIPTOR'

== two-drives
# measured in MAME: a new file goes onto the last drive that has a disk (drive 1), also when drive 0 has more room
-- text
x
-- case
terminal: console
device: drive0 disk0.img
device: drive1 disk1.img
device: ud
host: python3 $WEGA_HOST/tools/udosfs.py format disk0.img && python3 $WEGA_HOST/tools/udosfs.py format disk1.img
step: putud text
host: python3 $WEGA_HOST/tools/udosfs.py ls disk1.img | grep -q '^text ' && echo on-drive-1
host: python3 $WEGA_HOST/tools/udosfs.py ls disk0.img | grep -q '^text ' || echo not-on-drive-0
expect: output contains 'on-drive-1'
expect: output contains 'not-on-drive-0'

# ---- the signals (src: main, del_pro) -----------------------------------------------------------------------------
# putud catches INT (del_pro: caught again, the flag del set); HUP, QUIT, ILL, TRAP, IOT, SEGV, SYS, PIPE, ALRM, TERM
# are ignored. `p` is a FIFO: putud waits in its open until the signal comes.

== signal-int
# src: INT while p is opened: the open ends with EINTR ("p unable to open file"); del is set, so for the next file the
# answer to its first block is taken for ESC: CAN, the whole run ends (exit status 1: one file not sent); k: the 8-bit
# part has the block already and closes the file with it (text has its one block)
-- text
first line
second line
-- case
terminal: console
device: drive0 disk.img
device: ud
host: python3 $WEGA_HOST/tools/udosfs.py format disk.img && mkfifo p
step: putud p text
signal: INT after 2
host: python3 $WEGA_HOST/tools/udosfs.py ls disk.img
expect: output contains 'p unable to open file'
expect: output contains 'step 1 putud: exit=1'
expect: output contains 'text                             A0'

== signal-quit-ignored
# src: QUIT is ignored: putud goes on waiting in the open (the step ends by its TIMEOUT)
-- case
terminal: console

device: ud
host: python3 $WEGA_HOST/tools/udosfs.py format disk.img && mkfifo p
