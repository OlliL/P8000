# The cases of the corpus of bc (compare/bc/corpus.py); the format: host/recon/casefile.py (the default file is t.in, the
# default step bc t.in: bc reads t.in, then the standard input (empty), and ends dc with q).
# Where an expectation comes from: man = /usr/man/man1/bc.1, src = src/cmd/bc/bc.y (rule of the grammar or function),
# dc = the arithmetic of dc(1) (results truncated to scale digits).

== man-example
# man: BEISPIELE, the exponential function, here for 0 and 1 with scale 5; dc: every term a/b truncated to 5 digits
# (1+1+.5+.16666+.04166+.00833+.00138+.00019+.00002)
scale = 5
define e(x){
	auto a, b, c, i, s
	a = 1
	b = 1
	s = 1
	for(i=1; 1==1; i++){
		a = a*x
		b = b*i
		c = a/b
		if(c == 0) return(s)
		s = s+c
	}
}
e(0)
e(1)
-- expect
1
2.71824
