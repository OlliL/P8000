# The cases of the corpus of ps (compare/ps/corpus.py); the format: host/recon/casefile.py (the default step is ps).
# Where an expectation comes from: src = src/cmd/ps/ps.c (function), kva.c. The dump of the -k cases is made by
# compare/ps/mkdump.py (its text names the processes); the clock stands at 425000000 (STIME).
== usage
# src main: an unknown option: getopt complains, then the usage (two lines) on stderr, exit 1
-- case
time: 425000000
step: ps -x
expect: exit 1
expect: output contains 'usage: ps [ -edalf ]'


== extra
# src main: an argument that is no option: the usage, exit 1
-- case
time: 425000000
step: ps extra
expect: exit 1
expect: output contains '[ -g glist ] [-k kcore]'


== nonamelist
# src main: nlist finds none of _proc, _swplo, _Nproc: "ps: no namelist", exit 1
-- case
time: 425000000
step: ps -n /nix
expect: exit 1
expect: output contains 'ps: no namelist'


== nomem
# src main: /dev/kmem cannot be opened (the emulator has none): "ps: no mem", exit 1
-- case
time: 425000000
step: ps -e
expect: exit 1
expect: output contains 'ps: no mem'


== nocore
# src main: -k with a file that does not exist: "ps: no mem", exit 1
-- case
time: 425000000
step: ps -k /nix -e
expect: exit 1
expect: output contains 'ps: no mem'


== badcore
# kva.c kvtop: a file without an MMU table: the kernel address is not mapped: "ps: error on kvseek", exit 1
-- case
time: 425000000
host: printf "x" > core
step: ps -k core -e
expect: exit 1
expect: output contains 'ps: error on kvseek'


== every
# src main, prcom: -e: every process: PID TTY TIME COMMAND; the zombie as <defunct>; the swapped one with an empty user block
