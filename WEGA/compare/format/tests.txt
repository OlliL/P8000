# The cases of the corpus of format (compare/format/corpus.py); the format: host/recon/casefile.py. The disks are image
# files of the case (`d1`, `d0`); a disk is made by python3 (the bytes) or udosfs.py. The answer to "ready ?" is the file
# `y` or `n`. Where an expectation comes from: man = /usr/man/man1/format.1, src = src/cmd/format/1.5/format.c, k = the
# coupling software (UDOS/WEGA/K.BLK.S: request F).

== default
# man: format formats the floppy in drive 1 (/dev/rfd1: 96ds9); src: the question on the console, y; k: every track
# formatted (E5H) and read back, ACK; src: a bell, exit status 0
-- y
y
-- case
terminal: console
device: drive1 d1
device: ud
host: python3 -c "open('d1','wb').write(b'x' * 1000)"
step: format
stdin: y
host: wc -c < d1; od -An -tx1 -N 4 d1
expect: output contains 'Format /dev/rfd1, ready ? '
expect: output contains '737280'
expect: output contains 'e5 e5 e5 e5'
expect: exit 0

== not-a-floppy
# src: a device of another driver (/dev/null: major 6) and a plain file are not floppy devices
-- case

step: format /etc/passwd
expect: output contains '/dev/null is not a floppy device'
