# The cases of the corpus of sysgen (compare/sysgen/corpus.py); the format: host/recon/casefile.py. Every case runs as root with the clock
# at 600000000. /usr/sys is a copy in the layer; the answers are the standard input (gets). After a run the kernel (wega.out), the boot
# program (boot.out) and the headers it wrote (headers.out) are copied to the case directory and compared. The link of the kernel ends
# with "Undefined: _cpu2_fl" in the emulator (make -i ignores it): the same on both sides. Where an expectation comes from: src =
# src/cmd/sysgen.sh (a csh script).

== nothing
# src: n to every question: the headers stay (mdsize.h, ver.h as the tree has them), the kernel and the boot program are built
-- a
n
n
n
n
-- case
step: /etc/sysgen
stdin: a
host: cd $Z8K_OVERLAY/usr/sys && for f in conf/wega conf/boot; do [ -f $f ] && cp $f $OLDPWD/$(basename $f).out; done; cat h/mdsize.h h/ver.h > $OLDPWD/headers.out
expect: exit 0
expect: output contains '---- sysgen ----'
expect: output contains '#define SYSNAME		"P8000"'
expect: output contains 'New wega is building now ! Wait a minute !'
expect: output contains 'New boot is building now !'
