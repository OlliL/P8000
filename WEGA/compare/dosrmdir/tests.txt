# The cases of the corpus of dosrmdir (compare/dosrmdir/corpus.py); the format: host/recon/casefile.py. Every case starts with a
# floppy in drive 1 formatted by the original dosformat (host/recon/dospre.py) and the guest clock at 600000000 (1989).
# The disk: 713 clusters of 1024 bytes, all free at the start. The directories and files are put there by the original
# dosmkdir and doscp; what a case reads back it reads with the original dosls, dosdir and host/tools/dcpimg.py (free
# clusters, the raw entries: the name of a deleted entry starts with E5). Where an expectation comes from: man =
# /usr/man/man1/dos.1, src = dosrmdir.c, dosdel.c.

== simple
# man: dosrmdir removes a directory of the DCP disk: the entry is marked deleted (E5), the cluster of the directory is free
-- case
step: /usr/bin/dosmkdir /dev/rfd1:d
host: echo free=$(python3 $WEGA_HOST/tools/dcpimg.py free d1)
step: dosrmdir /dev/rfd1:d
host: echo free=$(python3 $WEGA_HOST/tools/dcpimg.py free d1); python3 $WEGA_HOST/tools/dcpimg.py entries d1
step: /usr/bin/dosls /dev/rfd1:
expect: exit 0
expect: output contains 'free=712'
expect: output contains 'free=713'
expect: output contains "'\\xe5          ' 10 2 0"
expect: output is '\n/dev/rfd1:/\n'

== second-cluster
# src (a quirk, KNOWLEDGE.md): a directory in the second cluster of the directory above (30 files make the first one full):
# the search for the entry reads the first cluster again (nxt is 0 after the count of the clusters), finds nothing and
# marks as deleted the 32-byte entry that follows the cluster on the disk: the first byte of the data of the first file
# (cluster 3). The directory is still listed, its cluster is free, status 0.
-- case
host: for i in $(seq 1 30); do echo $i > g$i; done; touch -d '1989-03-05 08:07:06 UTC' g*
step: /usr/bin/dosmkdir /dev/rfd1:p
step: /usr/bin/doscp g1 g2 g3 g4 g5 g6 g7 g8 g9 g10 g11 g12 g13 g14 g15 g16 g17 g18 g19 g20 g21 g22 g23 g24 g25 g26 g27 g28 g29 g30 /dev/rfd1:p
host: echo g1=$(od -An -tx1 -j8192 -N3 d1) free=$(python3 $WEGA_HOST/tools/dcpimg.py free d1)
step: /usr/bin/dosmkdir /dev/rfd1:p/q
host: echo free=$(python3 $WEGA_HOST/tools/dcpimg.py free d1)
step: dosrmdir /dev/rfd1:p/q
host: echo g1=$(od -An -tx1 -j8192 -N3 d1) free=$(python3 $WEGA_HOST/tools/dcpimg.py free d1)
step: /usr/bin/dosdir /dev/rfd1:p
expect: exit 0
expect: output contains 'g1= 31 0d 0a free=682'
expect: output contains 'free=680'
expect: output contains 'g1= e5 0d 0a free=681'
expect: output contains 'Q             <DIR>'
