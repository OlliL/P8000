# The cases of the corpus of chkdiff (compare/chkdiff/corpus.py); the format: host/recon/casefile.py. Every case runs as root
# with the clock at 600000000. The versions are made by the original /bin/chkin and /bin/chkout: f has version 1.1 (one, two,
# three), 1.2 (one, TWO, three, four) and the working file is 1.2 plus a line five. Where an expectation comes from: man =
# /usr/man/man1/chkdiff.1, src = src/cmd/chkdiff.sh (a csh script, run by /bin/csh as its first byte is #).

== last
# man: without options the source file is compared with the last version in the control file f+: the line five is new
-- f
one
two
three
-- case
step: /bin/chkin -c first f
step: /bin/chkout -e f
host: chmod 644 f; printf 'one\nTWO\nthree\nfour\n' > f
step: /bin/chkin -c second f
host: chmod 644 f; printf 'one\nTWO\nthree\nfour\nfive\n' > f
step: chkdiff f
expect: exit 0
expect: output is '4a5\n> five\n'

== version
# man: -v rel.lev compares the source file with that version: 1.1 has two, three and no four, five
-- f
one
two
three
-- case

step: /bin/chkout -e f

step: /bin/chkin -c second f
host: chmod 644 f; printf 'one\nTWO\nthree\nfour\nfive\n' > f
step: chkdiff -v 1.1 f
expect: exit 0
expect: output is '2c2\n< two\n---\n> TWO\n3a4,5\n> four\n> five\n'

== subdirectory
# src (a quirk): the directory part is split off for the control file (d/f+ is read) but the source file is the plain name f in
# the current directory (diff - f)
-- d/f
one
-- case
step: /bin/chkin -d d -c first f
host: mv f+ d/f+; printf 'one\ntwo\n' > f
step: chkdiff d/f
expect: exit 0
expect: output is '1a2\n> two\n'
