# The cases of the corpus of pburn (compare/pburn/corpus.py: every block for board 1 and board 2); the format:
# host/recon/casefile.py. The answers of the dialogue are the file `a` (stdin). The EPROM is chip.bin; $MK makes its
# contents: python3 -c "..." (a pattern, or erased: FFH).
# Where an expectation comes from: man = /usr/man/man1/pburn.1, src = src/cmd/pburn/2.1, k = the coupling software
# (UDOS/WEGA/K.PROM.S, K.P8000.S: request P).

# ---- the public interface (man page) ----------------------------------------------------------------------------

== menu
# man: pburn shows the board it found and its menu; q leaves it
-- a
q
-- case
terminal: console
device: ud
device: eprom chip.bin e2716 board={board}
step: pburn
stdin: a
expect: output contains 'pburn - prom programmer board {board}'
expect: output contains 'q - quit'

== bytemode
# man: b, bytemode: one byte at a time; src (modprog): "address to program ?", the byte read from the EPROM, a new value
# is programmed, q ends; then the CRC. A bug of pburn: the value is read by sscanf("%x") into a char, the Z8000 is
# big-endian: the high byte of the int (00) lands in the char, the 55 behind it; 00 is programmed
-- a
b
e2716
y
10
55
q
q
q
-- case
terminal: console
device: ud
device: eprom chip.bin e2716 board={board}
step: pburn
stdin: a
host: od -An -tx1 -j 16 -N 2 chip.bin
expect: output contains 'address to program ?'
expect: output contains '0010:	ff'
expect: output contains ' 00 ff'
