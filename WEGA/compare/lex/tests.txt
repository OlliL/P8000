# The cases of the corpus of lex (compare/lex/corpus.py); the format: host/recon/casefile.py. src = src/cmd/lex (System III
# lex): lex [-tvnfcr] [file ...]; without -t the program goes to lex.yy.c, with -t to stdout; -v statistics (stderr),
# -n none; -f no packing (optim); -r ratfor (/usr/lib/lex/nrform, not in WEGA: the error of the missing form).

== words
%%
[a-z]+	printf("word %s\n", yytext);
[0-9]+	printf("number %s\n", yytext);
.	;
\n	ECHO;

== words-stdout
%%
[a-z]+	printf("word %s\n", yytext);
[0-9]+	printf("number %s\n", yytext);
-- case
step: lex -t t.in

== definitions
D	[0-9]
L	[a-zA-Z_]
%{
int count;
%}
%%
{L}({L}|{D})*	{ count++; printf("id %s\n", yytext); }
{D}+"."{D}*	printf("real\n");
"/*"	{ while (input() != '*') ; }
^#.*$	printf("directive\n");
[ \t]+$	;
%%
main() { yylex(); printf("%d\n", count); }

== start-conditions
%Start COMMENT STR
%%
"/*"	BEGIN COMMENT;
<COMMENT>"*/"	BEGIN 0;
<COMMENT>.|\n	;
\"	BEGIN STR;
<STR>\"	BEGIN 0;
<STR>[^"]*	printf("%s", yytext);
ab/cd	printf("ab before cd\n");
x{2,4}	printf("xs\n");
a?b*c+	printf("abc\n");
[^a-z\n]	printf("other\n");
(ab|cd)$	printf("end\n");
