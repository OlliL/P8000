# The cases of the corpus of lint (compare/lint/corpus.py); the format: host/recon/casefile.py. Where an expectation comes from: src =
# src/cmd/lint.sh (a sh script: for every argument that is not an option the file goes through /lib/cpp -C -Dlint and /usr/lib/lint1
# (-I, -D, -U for cpp; -l: a library /usr/lib/llib<name>.ln; -p, -n and other letters to lint1: -p: portability and llib-port.ln, -n: no
# library); at the end the library llib-lc.ln (or llib-port.ln, none with -n) and lint2 -T over all of it).

== warnings
-- a.c
#include <stdio.h>
int f(a, b)
int a;
{
	int unused;
	return a;
}
main()
{
	int x;
	x = f(1, 2);
	printf("%d\n", x);
	f(3, 4);
}
-- case
step: /usr/bin/lint a.c
expect: output contains 'unused unused in function f'
expect: output contains 'function returns value which is always ignored'
expect: output contains 'printf'

== library-lc
# src: uname is in llib-lc.ln (the library of the system) but not in llib-port.ln
-- u.c

{
	uname(0);
	return 0;
}
-- case
step: /usr/bin/lint u.c
expect: output contains 'llib-lc(172)'

main()
