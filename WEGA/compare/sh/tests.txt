== control
# xec.c: if, while, until, for, case, break, continue
-- case
step: sh t.in
expect: output is 'i=1\ni=3\nn=1\nn=2\nn=0\nabc a\nx12 digit\nzz other\n'
-- t.in
for i in 1 2 3 4
do
	if test $i = 2
	then continue
	elif test $i = 4
	then break
	fi
	echo i=$i
done
n=0
while test $n -lt 2; do n=`expr $n + 1`; echo n=$n; done
until test $n -eq 0; do n=`expr $n - 1`; done; echo n=$n
for w in abc x12 zz
do
	case $w in
	a*) echo $w a;;
	*[0-9]*) echo $w digit;;
	*) echo $w other;;
	esac
done

== parameters

-- case
