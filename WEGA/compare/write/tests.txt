# The cases of the corpus of write (compare/write/corpus.py); the format: host/recon/casefile.py. Every case has the guest
# clock at 600000000. /etc/utmp is made by the case in the layer over the tree (host/tools/mkutmp.py: LINE:NAME), the terminals of
# the other users are image files (device: ttyN image:tN), the own terminal is the console (terminal: console, ttyname
# /dev/console). Where an expectation comes from: man = /usr/man/man1/write.1, src = src/cmd/write.c.

== message
# man: write copies what is typed to the terminal of the user, after a head "Message from NAME TTY..." and three bells;
# at the end of the input the other side gets EOF
-- hi
hello there
-- case
terminal: console
host: : > t2; mkdir -p $Z8K_OVERLAY/etc && python3 $WEGA_HOST/tools/mkutmp.py $Z8K_OVERLAY/etc/utmp console:me tty2:bob
device: tty2 image:t2
step: write bob
stdin: hi
host: cat -v t2
expect: exit 0
expect: output contains 'Message from me console...'
expect: output contains '^G^G^Ghello there'
expect: output contains 'EOF'

== twice

== shell
# man: a message that begins with ! is given to the shell (the whole block that was read: one read of up to 128 bytes);
# its output stays on the own terminal, then "!" is printed
-- in
!echo from the shell
-- case
terminal: console
host: : > t2; mkdir -p $Z8K_OVERLAY/etc && python3 $WEGA_HOST/tools/mkutmp.py $Z8K_OVERLAY/etc/utmp console:me tty2:bob
device: tty2 image:t2
step: write bob
stdin: in
expect: exit 0
expect: output is 'from the shell\n!\n'
